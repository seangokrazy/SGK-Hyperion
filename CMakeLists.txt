cmake_minimum_required(VERSION 3.16)
project(SGKHyperion VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(hyperion_core STATIC Source/DSP/BankLoader.cpp Source/DSP/PcmDecoder.cpp)
target_include_directories(hyperion_core PUBLIC Source/DSP)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|ARM)")
  target_compile_definitions(hyperion_core PUBLIC SGK_HYPERION_FAST_SINE=1)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  add_library(sgk_hyperion SHARED Source/MPC/Plugin.cpp)
  target_link_libraries(sgk_hyperion PRIVATE hyperion_core Threads::Threads ${CMAKE_DL_LIBS})
  set_target_properties(sgk_hyperion PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
endif()
add_executable(hyperion-catalog Tools/build_catalog.cpp)
target_link_libraries(hyperion-catalog PRIVATE hyperion_core)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  foreach(test AnalogTests FmTests WaveFxTests FxEditTests FastSineTests DecoderTests DspTests InstrumentTests)
    add_executable(${test} Tests/${test}.cpp)
    target_link_libraries(${test} PRIVATE hyperion_core)
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
endif()
